# Integration test requirements for hermes-tray.
#
# These tests hit a running hermes-agent-cn gateway (default 127.0.0.1:8642)
# through plain HTTP.  They are NOT part of the bundled Tauri app, so the
# runtime requirements live here rather than in the top-level package.json.
#
# Install into a virtualenv (any Python 3.10+ works):
#
#     python -m venv .venv-tests
#     .venv-tests\Scripts\activate          # Windows
#     source .venv-tests/bin/activate       # WSL / Linux / macOS
#     pip install -r requirements.txt
#
pytest>=7.0
httpx>=0.27
# `openai` is used to verify the gateway is wire-compatible with the
# official OpenAI Python SDK.  hermes-tray doesn't ship with it (it uses
# fetch), but the integration contract claims OpenAI-shape, so the suite
# pulls it in to prove the claim.  Any 1.x or 2.x release works.
openai>=1.0
# `httpx-sse` is optional — the test suite has its own SSE parser because
# hermes-agent-cn emits non-spec `event: hermes.tool.progress` lines
# interleaved with the spec `data:` lines, and we want to assert the raw
# shape rather than rely on a higher-level abstraction.  Kept here as a
# hint for future contributors who want a quick out-of-the-box SSE
# parser in their own tests.
httpx-sse>=0.4
